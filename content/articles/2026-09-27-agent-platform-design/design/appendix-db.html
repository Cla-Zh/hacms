PostgreSQL 16 + Redis 7

平台 S 端的完整数据模型。8 张 PG 表 + 12 类 Redis 数据结构。涵盖用户、VM、挂载、运行时、对话、用量、审计、JWT 撤销。配套 Alembic 迁移、索引策略、备份与归档方案。

PG 16 8 张表 Redis 7 12 类 key 迁移 Alembic 备份 pg_dump 每日 归档 S3 [v0.3]

B.1数据模型总览

设计原则

  • PG 是 SoT:所有关系数据 + 审计 + 配置 → PostgreSQL 16
  • Redis 是缓冲:会话/速率限制/事件流/JWT 撤销表 → Redis 7
  • JSONB 优先:半结构化数据 (manifest, config) 用 JSONB 而非拆表
  • 软删除:所有业务表有 deleted_at TIMESTAMPTZ NULL, 90 天后真删
  • 审计触发器:每个表变更自动写 audit_logs 表

B.3PostgreSQL 表结构 (8 张)

B.3.1 users

CREATE TYPE user_role AS ENUM ('user', 'admin');

CREATE TABLE users (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email         CITEXT UNIQUE NOT NULL,
  display_name  VARCHAR(64) NOT NULL DEFAULT '',
  role          user_role NOT NULL DEFAULT 'user',
  preferences   JSONB NOT NULL DEFAULT '{}'::jsonb,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at    TIMESTAMPTZ
);

CREATE INDEX idx_users_email ON users(email) WHERE deleted_at IS NULL;

B.3.2 llm_api_keys

CREATE TYPE llm_provider AS ENUM ('openai', 'anthropic', 'google', 'mistral', 'custom');

CREATE TABLE llm_api_keys (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  provider        llm_provider NOT NULL,
  encrypted_key   BYTEA NOT NULL,         -- AES-256-GCM 加密
  fingerprint     VARCHAR(16) NOT NULL,    -- sha256[:8]
  label           VARCHAR(64),
  last_used_at    TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ,
  UNIQUE(user_id, provider, fingerprint)
);

CREATE INDEX idx_llm_keys_user ON llm_api_keys(user_id) WHERE deleted_at IS NULL;

B.3.3 runtimes

CREATE TABLE runtimes (
  id            VARCHAR(64) PRIMARY KEY,   -- e.g. "rt_openclaw_030a6ec"
  name          VARCHAR(64) NOT NULL,
  version       VARCHAR(32) NOT NULL,
  source        VARCHAR(255) NOT NULL,    -- github://..., oci://...
  is_builtin    BOOLEAN NOT NULL DEFAULT FALSE,
  owner_id      UUID REFERENCES users(id) ON DELETE SET NULL,
  manifest      JSONB NOT NULL,            -- 完整 manifest.json
  sha256        BYTEA NOT NULL,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at    TIMESTAMPTZ
);

CREATE INDEX idx_runtimes_owner ON runtimes(owner_id) WHERE deleted_at IS NULL;

B.3.4 vms (核心表)

CREATE TYPE vm_state AS ENUM (
  'NEW', 'STARTING', 'RUNNING', 'PAUSED', 'STOPPING', 'STOPPED', 'FAILED', 'DESTROYED'
);

CREATE TABLE vms (
  id              VARCHAR(64) PRIMARY KEY,   -- e.g. "vm_abc123def"
  owner_id        UUID NOT NULL REFERENCES users(id),
  runtime_id      VARCHAR(64) NOT NULL REFERENCES runtimes(id),
  name            VARCHAR(64) NOT NULL,
  state           vm_state NOT NULL DEFAULT 'NEW',
  config          JSONB NOT NULL,            -- 完整 VM 配置
  pid             INTEGER,                   -- S 端 VM Manager 子进程 PID
  started_at      TIMESTAMPTZ,
  paused_at       TIMESTAMPTZ,
  last_active_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ
);

CREATE INDEX idx_vms_owner_state ON vms(owner_id, state) WHERE deleted_at IS NULL;
CREATE INDEX idx_vms_runtime ON vms(runtime_id);
CREATE INDEX idx_vms_last_active ON vms(last_active_at);  -- 用于 idle killer

B.3.5 vm_mounts

CREATE TYPE mount_mode AS ENUM ('read-only', 'read-write');
CREATE TYPE mount_state AS ENUM ('PENDING', 'MOUNTED', 'DETACHED', 'QUOTA_EXCEEDED', 'ERROR');

CREATE TABLE vm_mounts (
  id              VARCHAR(64) PRIMARY KEY,
  vm_id           VARCHAR(64) NOT NULL REFERENCES vms(id) ON DELETE CASCADE,
  host_path       TEXT NOT NULL,
  vm_path         TEXT NOT NULL,
  mode            mount_mode NOT NULL DEFAULT 'read-write',
  quota_mb        INTEGER,
  state           mount_state NOT NULL DEFAULT 'PENDING',
  io_read_bytes   BIGINT NOT NULL DEFAULT 0,
  io_write_bytes  BIGINT NOT NULL DEFAULT 0,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ,
  UNIQUE(vm_id, vm_path)
);

CREATE INDEX idx_mounts_vm ON vm_mounts(vm_id) WHERE deleted_at IS NULL;

B.3.6 conversations

CREATE TABLE conversations (
  id              VARCHAR(64) PRIMARY KEY,
  vm_id           VARCHAR(64) NOT NULL REFERENCES vms(id) ON DELETE CASCADE,
  title           VARCHAR(255) NOT NULL DEFAULT '',
  summary         TEXT,
  message_count   INTEGER NOT NULL DEFAULT 0,
  tokens_in       BIGINT NOT NULL DEFAULT 0,
  tokens_out      BIGINT NOT NULL DEFAULT 0,
  last_msg_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ
);

CREATE INDEX idx_conversations_vm ON conversations(vm_id, last_msg_at DESC) WHERE deleted_at IS NULL;

B.3.7 messages

CREATE TYPE msg_role AS ENUM ('user', 'assistant', 'system', 'tool');

CREATE TABLE messages (
  id              BIGSERIAL PRIMARY KEY,
  conv_id         VARCHAR(64) NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
  seq             INTEGER NOT NULL,
  role            msg_role NOT NULL,
  content         TEXT NOT NULL,
  tool_calls      JSONB,                   -- [{name, args, result, duration_ms}]
  tokens_in       INTEGER NOT NULL DEFAULT 0,
  tokens_out      INTEGER NOT NULL DEFAULT 0,
  duration_ms     INTEGER,
  trace_id        VARCHAR(32),             -- OTel trace ID
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE(conv_id, seq)
);

CREATE INDEX idx_messages_conv ON messages(conv_id, seq);
-- 时序优化: BRIN 索引 (按时间)
CREATE INDEX idx_messages_created_brin ON messages USING BRIN(created_at);

B.3.8 audit_logs

CREATE TYPE actor_type AS ENUM ('user', 'system', 'admin', 'scheduler');

CREATE TABLE audit_logs (
  id              BIGSERIAL PRIMARY KEY,
  ts              TIMESTAMPTZ NOT NULL DEFAULT now(),
  actor_id        UUID,
  actor_type      actor_type NOT NULL,
  action          VARCHAR(64) NOT NULL,    -- vm.create / mount.add / ...
  resource_type   VARCHAR(32) NOT NULL,
  resource_id     VARCHAR(64),
  ip              INET,
  ua              VARCHAR(255),
  payload         JSONB,
  prev_hash       BYTEA NOT NULL,           -- 前一行 hash (链式)
  hash            BYTEA NOT NULL
);

CREATE INDEX idx_audit_ts ON audit_logs(ts DESC);
CREATE INDEX idx_audit_actor ON audit_logs(actor_id, ts DESC);
CREATE INDEX idx_audit_action ON audit_logs(action, ts DESC);
CREATE INDEX idx_audit_resource ON audit_logs(resource_type, resource_id);

B.3.9 触发器:自动审计

-- 通用审计触发器函数
CREATE OR REPLACE FUNCTION audit_trigger() RETURNS TRIGGER AS $$
DECLARE
  v_actor_id UUID;
  v_actor_type actor_type;
  v_prev_hash BYTEA;
BEGIN
  -- 从 session 变量取 actor (S 端连接时设置 SET LOCAL app.actor_id = ...)
  v_actor_id := current_setting('app.actor_id', true)::UUID;
  v_actor_type := COALESCE(current_setting('app.actor_type', true), 'system')::actor_type;

  -- 取上一行 hash
  SELECT hash INTO v_prev_hash FROM audit_logs ORDER BY id DESC LIMIT 1;

  INSERT INTO audit_logs (actor_id, actor_type, action, resource_type, resource_id, payload, prev_hash, hash)
  VALUES (
    v_actor_id,
    v_actor_type,
    TG_OP || '_' || TG_TABLE_NAME,
    TG_TABLE_NAME,
    COALESCE((NEW.id)::text, (OLD.id)::text),
    jsonb_build_object('old', to_jsonb(OLD), 'new', to_jsonb(NEW)),
    v_prev_hash,
    digest(COALESCE(v_prev_hash, '\x'::bytea) || NEW::text::bytea, 'sha256')
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 绑定到 vms 表
CREATE TRIGGER audit_vms AFTER INSERT OR UPDATE OR DELETE ON vms
  FOR EACH ROW EXECUTE FUNCTION audit_trigger();

B.4索引策略

B.4.1 索引设计原则

  • 外键必建索引:所有 FK 列都有 B-tree 索引
  • 软删除过滤:几乎所有查询都加 WHERE deleted_at IS NULL,索引也带此谓词 (PG partial index)
  • 复合索引顺序:等值列在前,范围列在后;按 cardinality 降序
  • 时序数据用 BRIN:messages 表按时间排序,BRIN 索引体积小

B.5Redis 数据结构 (12 类 key)

Key 模式类型TTL用途
session:{jwt_id}Hash15minJWT 会话状态 (user_id, issued_at, ua, ip)
jwt:revoked:{jti}String15minJWT 撤销表 (登出 / 密码修改)
ratelimit:{actor}:{bucket}String (counter)60s速率限制 token bucket
magic_link:{token_hash}String15minMagic Link 兑换 (一次性)
vm:state:{vm_id}Hash—VM 实时状态缓存 (state, pid, ip, port)
vm:metrics:{vm_id}Stream24hVM 指标时间序列 (cpu, mem)
vm:events:{vm_id}Stream7dVM 事件日志 (启动/停止/工具调用)
ws:conn:{conn_id}Hash5min 心跳WebSocket 连接状态
ws:topic:{vm_id}:subscribersSet—订阅某 VM 的 WS 连接集合
tool:approval:{tool_call_id}Hash30s工具调用等待用户审批 (pending decision)
cache:vm_list:{user_id}JSON60sVM 列表 API 缓存
lock:vm:{vm_id}String (NX EX)5sVM 操作分布式锁 (避免双启动)

TTL 与淘汰策略

  • maxmemory-policy: allkeys-lru
  • 长 TTL key (vm:state) 不会被淘汰 (因为命中率低,影响小)
  • 短 TTL key (ratelimit / magic_link) 触发自动清理

B.6.3 迁移规范

  • 所有迁移必须 backward-compatible:新代码可读旧 schema
  • 大数据表迁移:用 CREATE INDEX CONCURRENTLY (不在事务中)
  • 字段重命名:旧字段保留 1 个版本,新字段加默认值 NULL,代码双写
  • 删除字段:标记 deprecated 1 个版本后才真删

B.8备份与恢复

B.8.1 备份策略

#!/bin/bash
# /opt/agent-platform/scripts/backup-pg.sh
# 每日 02:00 crontab 运行
TIMESTAMP=$(date +%Y%m%d-%H%M%S)
BACKUP_DIR=/var/backups/agent-platform/pg
mkdir -p "$BACKUP_DIR"

pg_dump \
  -h localhost -p 5432 -U agent_platform \
  -Fc --no-owner --no-acl \
  -d agent_platform \
  -f "$BACKUP_DIR/pg-$TIMESTAMP.dump"

# 压缩 (可选,custom format 已压缩)
# gzip "$BACKUP_DIR/pg-$TIMESTAMP.dump"

# 清理 30 天前的备份
find "$BACKUP_DIR" -name "pg-*.dump" -mtime +30 -delete

# 上传到 S3 [v0.3]
# aws s3 cp "$BACKUP_DIR/pg-$TIMESTAMP.dump" \
#   s3://agent-platform-backups/pg/ \
#   --storage-class STANDARD_IA

B.8.2 恢复演练

# 1. 停止 S 端 API
systemctl stop agent-platform-api

# 2. 恢复 PG
pg_restore -h localhost -U agent_platform -d agent_platform \
  --clean --if-exists \
  /var/backups/agent-platform/pg/pg-20260929-020000.dump

# 3. 启动 S 端 API
systemctl start agent-platform-api

# 4. 验证
curl http://localhost:8000/v1/vms -H "Authorization: Bearer $TOKEN"
